	\section{Introduction}
	\begin{frame}{Introduction}
		\begin{itemize}
			\item Dans le cadre du projet de programmation impérative, nous avons réalisé une implémentation de l'algorithme des K plus proches voisins (KPPV).
			\item L'objectif de ce projet était de réaliser un programme complet permettant d'exécuter l'algorithme des KPPV avec deux types de structures de données.
			\item Nous avons utilisé pour cela le langage C, la bibliothèque MLV, Python et Bash.
		\end{itemize}
	\end{frame}

	\section{Architecture du code}
	\begin{frame}{Organisation des modules}
		\begin{columns}[T]
			\column{0.5\textwidth}
				\textbf{Structures de données}
				\begin{itemize}\small
					\item \textbf{point.c} : Point multidimensionnel classifié.
					\item \textbf{vector.c} : Vecteur dynamique générique.
					\item \textbf{stack.c} : Pile générique.
					\item \textbf{point\_distance.c} : Association point/distance.
					\item \textbf{ordered\_points\_list.c} : Liste ordonnée des k plus proches voisins.
					\item \textbf{points\_list.c} : Liste de points (force brute).
				\end{itemize}
			\column{0.5\textwidth}
				\textbf{Algorithmes et interface}
				\begin{itemize}\small
					\item \textbf{tree\_kd.c} : Recherche KPPV par arbre KD.
					\item \textbf{save.c} : Lecture/écriture des données.
					\item \textbf{generate.c} : Génération de points aléatoires.
					\item \textbf{interface.c} : Affichage graphique (MLV).
					\item \textbf{main.c} : Point d'entrée et mesures de performance.
					\item \textbf{test.sh / graphique.py} : Tests et graphiques.
				\end{itemize}
		\end{columns}
	\end{frame}
